package part1;


public class FractionalKnapsack {

    public static void main(String[] args) {

        int[] weight = {10, 20, 30};
        int[] profit = {60, 100, 120};

        int capacity = 50;
        int n = weight.length;

        double totalProfit = 0;

        // Calculate profit/weight ratio
        double[] ratio = new double[n];

        for (int i = 0; i < n; i++) {
            ratio[i] = (double) profit[i] / weight[i];
        }

        // Sort according to ratio in descending order
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {

                if (ratio[i] < ratio[j]) {

                    double temp = ratio[i];
                    ratio[i] = ratio[j];
                    ratio[j] = temp;

                    int t = weight[i];
                    weight[i] = weight[j];
                    weight[j] = t;

                    t = profit[i];
                    profit[i] = profit[j];
                    profit[j] = t;
                }
            }
        }

        // Fill the knapsack
        for (int i = 0; i < n; i++) {

            if (weight[i] <= capacity) {

                totalProfit += profit[i];
                capacity -= weight[i];

            } else {

                totalProfit += ratio[i] * capacity;
                capacity = 0;
                break;
            }
        }

        System.out.println("Maximum Profit: " + totalProfit);
    }
}

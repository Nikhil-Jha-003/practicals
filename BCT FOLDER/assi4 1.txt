// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract Student {

    // Structure
    struct StudentData {
        uint id;
        string name;
        uint age;
        string course;
    }

    // Array
    StudentData[] public students;

    // Add student details
    function addStudent(
        uint _id,
        string memory _name,
        uint _age,
        string memory _course
    ) public {
        students.push(
            StudentData(_id, _name, _age, _course)
        );
    }

    // Get student details
    function getStudent(uint index)
        public view returns (
            uint, string memory, uint, string memory
        )
    {
        StudentData memory s = students[index];

        return (s.id, s.name, s.age, s.course);
    }

    // Get total students
    function getStudentCount() public view returns(uint) {
        return students.length;
    }

    // Fallback function
    fallback() external payable {
    }
}

Step 3: Compile the Program
1. Click on Solidity Compiler.
2. Select compiler version 0.8.20 or compatible version.
3. Click on Compile Student.sol.
4. Wait for the green tick.
Step 4: Connect MetaMask
1. Open MetaMask extension.
2. Unlock your wallet.
3. Select Sepolia Test Network.
4. Return to Remix IDE.
5. Click on Deploy & Run Transactions.
6. Select Browser Extension in Environment.
7. Connect your MetaMask wallet.
Step 5: Deploy Smart Contract
1. Select the Student contract.
2. Keep Value as 0.
3. Click on Deploy.
4. MetaMask confirmation will appear.
5. Click on Confirm.
6. Wait for deployment to complete.
7. Open Deployed Contracts.
8. Expand the Student contract.
